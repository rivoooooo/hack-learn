### docker — 容器管理

**常用度**：★★★★★ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：非 POSIX

```bash
docker [全局选项] 子命令 [参数]
```

| 子命令组 | 常用命令 |
|----------|----------|
| 容器 | `run` `ps` `start` `stop` `restart` `rm` `logs` `exec` `inspect` `stats` `top` ⭐ |
| 镜像 | `images` `pull` `push` `build` `tag` `rmi` `history` `save` `load` ⭐ |
| 网络 | `network ls/create/inspect/rm` ⭐ |
| 卷 | `volume ls/create/inspect/rm/prune` ⭐ |
| 系统 | `info` `version` `system df` `system prune` ⭐ |
| 其他 | `compose` `context` `login` `exec` ⭐ |

```bash
# 场景：⭐⭐ 生命周期
$ docker run -d --name web -p 8080:80 nginx:1.24
$ docker ps                                     # 运行中的容器
$ docker ps -a                                  # 含已退出的 ⭐
$ docker logs -f web                            # 跟随日志 ⭐
$ docker exec -it web bash                      # 进入容器 ⭐
$ docker exec web nginx -t                       # ⭐ 不进入直接执行
$ docker stop web && docker rm web
$ docker rm -f web                               # ⚠️ 强制删除（含运行中）

# 场景：⭐⭐ run 的关键选项
$ docker run -d \
>   --name app \
>   -p 127.0.0.1:8080:80 \                        # ⭐ 只绑本机（更安全）
>   -v /srv/data:/data:ro \                       # ⭐ 只读挂载
>   -v app-data:/var/lib/app \                    # 命名卷（持久化）⭐
>   -e TZ=Asia/Shanghai -e LOG_LEVEL=info \       # ⭐ 环境变量
>   --env-file /etc/app.env \                     # ⭐ 从文件读环境变量
>   --restart unless-stopped \                    # ⭐ 自动重启
>   --memory 512m --cpus 1.5 --pids-limit 200 \   # ⭐ 资源限制
>   --read-only --tmpfs /tmp \                    # ⭐ 只读根 + 可写 /tmp
>   --cap-drop ALL --cap-add NET_BIND_SERVICE \   # ⭐🔐 最小 capability
>   --security-opt no-new-privileges \            # ⭐🔐
>   --user 1000:1000 \                            # ⭐ 非 root 运行
>   --health-cmd 'curl -fs localhost/health' --health-interval 30s \
>   myapp:1.0

$ # ⭐ 一次性容器（用完即删）
$ docker run --rm -it alpine sh
$ docker run --rm -v "$PWD":/w -w /w gcc:14 gcc -o out out.c

# 场景：⭐⭐ 镜像管理
$ docker images                                  # 本地镜像
$ docker image ls --format 'table {{.Repository}}\t{{.Tag}}\t{{.Size}}'   # ⭐
$ docker pull nginx:1.24-alpine                  # ⭐ 用具体版本（不用 latest）
$ docker tag myapp:1.0 registry.example.com/app:1.0
$ docker push registry.example.com/app:1.0
$ docker rmi nginx:1.24
$ docker image prune -a                          # ⚠️ 删所有未使用镜像
$ docker history nginx:1.24                      # ⭐ 看镜像分层历史
$ docker image inspect nginx:1.24 | jq '.[0].Config'   # ⭐

$ # ⭐ 离线搬运镜像
$ docker save myapp:1.0 | zstd -T0 > /tmp/myapp.tar.zst
$ zstd -dc /tmp/myapp.tar.zst | docker load
$ docker save -o /tmp/img.tar myapp:1.0

# 场景：⭐⭐ Dockerfile 最小可用模板
$ cat > /tmp/Dockerfile <<'DOCKER'
# 多阶段构建（最终镜像小）⭐
FROM golang:1.23-alpine AS build
WORKDIR /src
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN CGO_ENABLED=0 go build -ldflags '-s -w' -o /app ./cmd/app

FROM alpine:3.20
RUN apk add --no-cache ca-certificates tzdata && adduser -D -u 1000 app
COPY --from=build /app /usr/local/bin/app
USER app
EXPOSE 8080
HEALTHCHECK --interval=30s --timeout=3s CMD wget -qO- http://127.0.0.1:8080/health || exit 1
ENTRYPOINT ["/usr/local/bin/app"]
DOCKER
$ docker build -t myapp:1.0 /tmp
$ docker build -t myapp:1.0 --build-arg VERSION=1.0 -f /tmp/Dockerfile .
$ docker build --no-cache -t myapp:1.0 .          # ⚠️ 不用缓存

# 场景：⭐⭐ 清理（回收磁盘）
$ docker system df                               # ⭐ 占用概览
$ docker system df -v                            # 详细信息
$ docker container prune                         # 删已退出容器
$ docker image prune                             # 删悬空镜像
$ docker volume prune                            # ⚠️ 删未使用卷（数据会丢）
$ docker network prune
$ docker system prune -a --volumes               # ⚠️⚠️ 大扫除（可能删有用数据）
$ # ⭐ 只删 24 小时前的
$ docker container prune --filter 'until=24h'

# 场景：⭐⭐ 网络
$ docker network ls
$ docker network create --driver bridge mynet
$ docker network inspect mynet | jq '.[0].Containers'   # ⭐ 看哪些容器连着
$ docker run -d --network mynet --name db postgres
$ docker exec app ping -c1 db                     # ⭐ 同网络内按名字解析
$ docker network rm mynet
$ # ⭐ host 网络（无隔离，慎用）
$ docker run --rm --network host nginx            # ⚠️ 直接占用宿主端口

# 场景：⭐ 卷与数据
$ docker volume ls
$ docker volume create app-data
$ docker volume inspect app-data | jq -r '.[0].Mountpoint'   # ⭐ 宿主上的实际路径
$ docker run --rm -v app-data:/data alpine sh -c 'ls -la /data'
$ # ⭐ 备份卷
$ docker run --rm -v app-data:/data -v "$PWD":/backup alpine \
>   tar czf /backup/app-data.tar.gz -C /data .
$ # ⭐ 恢复
$ docker run --rm -v app-data:/data -v "$PWD":/backup alpine \
>   sh -c 'rm -rf /data/* && tar xzf /backup/app-data.tar.gz -C /data'

# 场景：⭐⭐ 排查（容器起不来 / 行为异常）
$ docker ps -a --filter 'status=exited'          # ⭐ 找退出容器
$ docker inspect app | jq '.[0].State'           # ⭐ 退出码与 OOM 信息
$ docker inspect app | jq -r '.[0].State.ExitCode, .[0].State.Error'
$ docker logs --tail 100 --timestamps app
$ docker events --since 1h                       # ⭐ 事件流
$ docker stats --no-stream                       # ⭐ 资源占用
$ docker top app                                 # 容器内进程（宿主视角）
$ docker diff app                                # ⭐ 文件系统改动（找被写的内容）
$ docker exec app cat /proc/1/cgroup             # ⭐ 确认 cgroup 限制
$ docker inspect -f '{{.State.OOMKilled}}' app   # ⭐ 是否被 OOM 杀掉

$ # ⭐ 容器内看真实资源限制
$ docker run --rm --memory 256m -it alpine sh -c \
>   'cat /sys/fs/cgroup/memory.max; cat /sys/fs/cgroup/cpu.max'

# 场景：🔐⭐⭐ 安全审计（特权与危险配置）
$ cat > /tmp/docker-audit.sh <<'SH'
#!/bin/bash
echo "=== 1. 特权容器（⚠️ 接近宿主 root）==="
docker ps -q | while read -r id; do
  priv=$(docker inspect -f '{{.HostConfig.Privileged}}' "$id")
  [ "$priv" = "true" ] && echo "⚠️ 特权容器: $(docker inspect -f '{{.Name}}' "$id")"
done

echo "=== 2. 挂载了 docker.sock 的容器（⚠️⚠️ 等于宿主 root）==="
docker ps -q | while read -r id; do
  docker inspect -f '{{.Name}} {{range .Mounts}}{{.Source}}->{{.Destination}} {{end}}' "$id" \
    | grep -q 'docker.sock' && echo "⚠️⚠️ $(docker inspect -f '{{.Name}}' "$id") 挂载了 docker.sock"
done

echo "=== 3. host 网络 / host PID / host IPC ==="
docker ps -q | while read -r id; do
  n=$(docker inspect -f '{{.Name}}' "$id")
  docker inspect -f 'net={{.HostConfig.NetworkMode}} pid={{.HostConfig.PidMode}} ipc={{.HostConfig.IpcMode}}' "$id" \
    | grep -qE 'net=host|pid=host|ipc=host' && echo "⚠️ $n $(docker inspect -f 'net={{.HostConfig.NetworkMode}} pid={{.HostConfig.PidMode}} ipc={{.HostConfig.IpcMode}}' "$id")"
done

echo "=== 4. 加了危险 capability 的容器 ==="
docker ps -q | while read -r id; do
  caps=$(docker inspect -f '{{range .HostConfig.CapAdd}}{{.}} {{end}}' "$id")
  [ -n "$caps" ] && echo "$(docker inspect -f '{{.Name}}' "$id"): +$caps"
done

echo "=== 5. 未限制资源的容器 ==="
docker ps -q | while read -r id; do
  m=$(docker inspect -f '{{.HostConfig.Memory}}' "$id")
  [ "$m" = "0" ] && echo "无内存限制: $(docker inspect -f '{{.Name}}' "$id")"
done

echo "=== 6. 以 root 运行的容器 ==="
docker ps -q | while read -r id; do
  u=$(docker inspect -f '{{.Config.User}}' "$id")
  [ -z "$u" ] || [ "$u" = "root" ] || [ "$u" = "0" ] && echo "root 用户: $(docker inspect -f '{{.Name}}' "$id")"
done

echo "=== 7. 敏感挂载（/、/etc、/root、宿主 /proc）==="
docker ps -q | while read -r id; do
  docker inspect -f '{{.Name}} {{range .Mounts}}{{.Source}}:{{.Destination}} {{end}}' "$id" \
    | grep -E ':/$|:/etc|:/root|:/proc|:/var/run' && echo "  ↑ 敏感挂载"
done

echo "=== 8. 所有镜像里以 root 运行的 ==="
docker images -q | sort -u | while read -r i; do
  docker image inspect "$i" -f '{{.RepoTags}} user={{.Config.User}}' 2>/dev/null
done | grep -E 'user=$|user=root|user=0' | head -20
SH
$ bash /tmp/docker-audit.sh

# 场景：🔐⭐ 从容器逃逸评估（在容器内部）
$ cat > /tmp/escape-check.sh <<'SH'
#!/bin/bash
echo "=== 1. 是否特权 ==="
capsh --print 2>/dev/null | grep -E 'Current|Bounding' | head -3
grep CapEff /proc/self/status

echo "=== 2. 危险的 capability ==="
capsh --print 2>/dev/null | grep -oE 'cap_sys_admin|cap_sys_ptrace|cap_sys_module|cap_dac_read_search|cap_net_admin' | sort -u

echo "=== 3. 可见的设备（特权容器的标志）==="
ls -la /dev/ | head -20

echo "=== 4. 挂载情况 ==="
mount | head -20
cat /proc/self/mountinfo | wc -l

echo "=== 5. 是否有 docker.sock / 宿主磁盘 ==="
ls -la /var/run/docker.sock /run/docker.sock 2>/dev/null
lsblk 2>/dev/null | head

echo "=== 6. namespace 隔离 ==="
ls -la /proc/self/ns/
echo "PID1: $(ps -p 1 -o comm= 2>/dev/null)"

echo "=== 7. seccomp ==="
grep Seccomp /proc/self/status

echo "=== 8. 容器标识 ==="
cat /proc/1/cgroup
ls -la /.dockerenv /run/.containerenv 2>/dev/null
cat /proc/1/environ 2>/dev/null | tr '\0' '\n' | grep -i container | head
SH
$ docker run --rm -v /tmp/escape-check.sh:/esc.sh:ro alpine sh /esc.sh

# 场景：🔐 用 docker 运行一个「逃逸演示」（仅授权环境）
```

#### 🔐 容器安全的四层隔离
| 层 | 不当配置的风险 |
|----|----------------|
| **namespace** | `--pid=host`（看/杀宿主进程）、`--network=host`（直接占用宿主网络）、`--privileged` 会共享几乎所有 namespace 🔐⚠️ |
| **cgroup** | 不限制资源 → 宿主机被拖垮 ⚠️ |
| **capability** | 默认有 ~14 个（含 `CAP_NET_RAW`）；`CAP_SYS_ADMIN` 几乎是完整的 root 🔐⚠️⚠️ |
| **seccomp/AppArmor/SELinux** | 关闭后系统调用无限制 🔐 |

**高危配置清单** ⭐⭐：
```
--privileged                     ⚠️⚠️ 接近宿主 root
-v /var/run/docker.sock:...      ⚠️⚠️ 等于宿主 root
--pid=host                       ⚠️ 可发信号给宿主进程
--network=host                   ⚠️ 网络无隔离
-v /:/host                       ⚠️⚠️ 宿主根文件系统
--cap-add SYS_ADMIN              ⚠️⚠️ 几乎等于 root
--cap-add SYS_PTRACE             ⚠️ 可注入其他进程
--security-opt seccomp=unconfined ⚠️ 关闭 seccomp
--security-opt apparmor=unconfined ⚠️ 关闭 AppArmor
--user root（默认）              ⚠️ 容器内是 root
--ipc=host                       ⚠️ 共享 IPC
```

#### 常见坑
- ⚠️⚠️ **`--privileged` 几乎等于把宿主 root 交出去**（可挂载宿主磁盘、改内核参数）🔐⭐⭐。
- ⚠️⚠️ **挂载 `docker.sock` 等于宿主 root**（容器内可用 docker CLI 执行任意容器）🔐⭐⭐。
- ⚠️⚠️ **`-p 8080:80` 默认绑 `0.0.0.0`**（暴露到公网）。**要只本机用 `-p 127.0.0.1:8080:80`** 🔐⭐⭐。
- ⚠️⚠️ **容器里的 `root` 不是宿主的 root**（除非配了 `--privileged`/`--userns=host`/大量 capability），但**默认仍是容器内的 root** ⭐。
- ⚠️⚠️ **`docker system prune -a --volumes` 会删除所有未使用的卷（数据丢失）** ⚠️⭐⭐。
- ⚠️ **`docker build` 的每一层都会被保存**（含被删的文件），**凭据写进 Dockerfile 会永久留在镜像层里** 🔐⭐⭐。
- ⚠️ **镜像用 `latest` 标签不可复现**（内容会变），生产用具体版本或 digest ⭐⭐。
- ⚠️ **`-v /host/path:/container/path` 是双向的**（容器可改宿主文件）。要只读加 `:ro` ⭐。
- ⚠️ **`docker exec -it` 需要容器里有 shell**（`scratch`/`distroless` 镜像没有）⭐：
  ```bash
  $ docker run --rm --pid=container:app --net=container:app -it alpine sh   # ⭐ 旁路注入
  ```
- ⚠️ **容器内的文件写入容器层会随删除消失**，要持久化用卷 ⭐。
- ⚠️ **`docker logs` 只显示 stdout/stderr**（容器内写文件看不到）⭐。
- **`docker` 需要守护进程（`dockerd`）与 root 或 `docker` 组** ⭐：
  ```bash
  $ ls -l /var/run/docker.sock
  srw-rw---- 1 root docker ... /var/run/docker.sock    # ⚠️ docker 组 = 实际 root
  ```
- 🔐 **加入 `docker` 组等于获得 root**（可挂载宿主 `/`）⭐。

### docker compose — 多容器编排

**常用度**：★★★★★ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：非 POSIX（v2 是 `docker compose`，v1 是 `docker-compose`）

```bash
docker compose [选项] 子命令
docker-compose [选项] 子命令      # v1（Python 版，已停止维护）
```

| 子命令 | 作用 |
|--------|------|
| `up -d` | **创建并后台启动全部服务** ⭐⭐ |
| `down` | **停止并删除容器/网络** ⭐⭐ |
| `ps` / `logs -f` | 状态 / 日志 ⭐ |
| `exec 服务 命令` | 在服务容器里执行命令 ⭐ |
| `run --rm 服务 命令` | 一次性运行（不占用服务容器） ⭐ |
| `build` | 构建镜像 ⭐ |
| `pull` / `push` | 拉取/推送镜像 ⭐ |
| `restart` / `stop` / `start` | 生命周期 ⭐ |
| `config` | **校验并打印合并后的配置** ⭐⭐ |

#### 常见坑
- ⚠️⚠️ **`docker compose down -v` 会删除命名卷（数据丢失）** ⚠️⭐⭐。
- ⚠️⚠️ **不固定容器名/端口时才能 `--scale`**（否则端口冲突）⭐。
- ⚠️⚠️ **`.env` 文件默认会被自动读取**（与 `--env-file` 不同），且**不要提交到 git** 🔐⭐⭐。
- ⚠️ **`depends_on` 只保证启动顺序，不保证服务就绪**。要等就绪用 `condition: service_healthy` ⭐⭐。
- ⚠️ **`compose.override.yml` 会自动叠加**，容易让人困惑「为什么配置和我写的不一样」⭐⭐：
  ```bash
  $ docker compose config      # ⭐ 看合并后的最终配置
  ```
- ⚠️ **`build: .` 的构建上下文是整个目录**（会把 `.env`、`.git` 也传进去）。用 `.dockerignore` 排除 ⭐⭐。
- ⚠️ **`latest` 标签不可复现**，生产用具体版本或 digest ⭐。
- **`docker compose` 是 v2（Go 插件），`docker-compose` 是 v1（Python，EOL）** ⭐。
- 🔐 **compose 文件常含明文凭据**（数据库密码、API Key）。**用环境变量 + 密钥管理** ⭐⭐。

### podman — 无守护进程容器

**常用度**：★★★★☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：非 POSIX（Red Hat 主导）

```bash
podman [全局选项] 子命令 [参数]
```

| 特点 | 说明 |
|------|------|
| **无守护进程** | 每个容器是调用者的子进程（无 `dockerd`） ⭐⭐ |
| **rootless** | 普通用户可运行容器（user namespace） ⭐⭐ |
| **兼容 Docker CLI** | `alias docker=podman` 基本可用 ⭐ |
| **支持 Pod** | 可管理 K8s 风格的 pod ⭐ |
| **systemd 集成** | `podman generate systemd` / Quadlet ⭐⭐ |

```bash
# 场景：⭐⭐ 基本用法（与 docker 几乎相同）
$ podman run -d --name web -p 8080:80 nginx:1.24-alpine
$ podman ps -a
$ podman exec -it web sh
$ podman logs web
$ podman rm -f web
$ podman build -t myapp:1.0 .
$ podman images

$ # ⭐ 兼容性：直接用 docker 语法
$ alias docker=podman
$ docker run --rm hello-world

# 场景：⭐⭐ rootless 容器（无需 root）
$ id -u                              # 非 0
$ podman run --rm -it alpine id
uid=0(root) gid=0(root)              # ⭐ 容器内是 root（映射到宿主普通用户）
$ cat /etc/subuid | grep "$USER"     # ⭐ 必需的 subuid/subgid 配置
$ cat /etc/subgid | grep "$USER"
$ podman unshare cat /proc/self/uid_map        # ⭐ 看 UID 映射
$ podman info --format '{{.Host.Security.Rootless}}'

$ # ⭐ 从 docker 迁移
$ podman run --privileged --rm docker.io/library/alpine sh -c 'echo ok'
$ podman images --format '{{.Repository}}:{{.Tag}}'
$ mv ~/.docker/config.json ~/.config/containers/auth.json 2>/dev/null

# 场景：⭐⭐ pod（K8s 风格）⭐
$ podman pod create --name mypod -p 8080:80
$ podman run -d --pod mypod --name web nginx
$ podman run -d --pod mypod --name sidecar alpine sleep 3600
$ podman pod ps
$ podman pod inspect mypod | jq '.InfraContainerID'
$ podman pod stop mypod && podman pod rm mypod

$ # ⭐ 生成 K8s YAML（从本地 pod 导出）⭐
$ podman generate kube mypod > /tmp/mypod.yaml
$ kubectl apply -f /tmp/mypod.yaml
$ podman play kube /tmp/mypod.yaml

# 场景：⭐⭐ systemd 集成（生产常用）⭐
$ # 方法 1：podman generate systemd（旧版）
$ podman generate systemd --new --name web --files
$ sudo mv container-web.service /etc/systemd/system/
$ sudo systemctl daemon-reload && sudo systemctl enable --now container-web

$ # 方法 2：Quadlet（podman 4.4+ 推荐）⭐
$ mkdir -p ~/.config/containers/systemd
$ cat > ~/.config/containers/systemd/app.container <<'UNIT'
[Unit]
Description=My App Container
After=network-online.target

[Container]
Image=myapp:1.0
PublishPort=127.0.0.1:8080:8080
Volume=app-data:/data
Environment=TZ=Asia/Shanghai
AutoUpdate=registry

[Service]
Restart=always

[Install]
WantedBy=default.target
UNIT
$ systemctl --user daemon-reload
$ systemctl --user start app
$ systemctl --user status app
$ systemctl --user enable app

$ # ⭐ 用户级容器在未登录时也运行
$ loginctl enable-linger "$USER"

# 场景：🔐 安全（podman 的优势）
$ podman info --format '{{.Host.Security.Rootless}}'      # true = rootless
$ podman inspect web --format '{{.HostConfig.Privileged}}'
$ podman inspect web --format '{{.HostConfig.CapDrop}}'
$ # ⭐ rootless 下即使容器被攻破，攻击者拿到的是宿主普通用户（不是 root）
$ # ⚠️ 但如果在容器内提权配合内核漏洞，仍可能逃逸

# 场景：🧪 podman vs docker 的差异
$ cat <<'EOF'
=== podman 与 docker 的关键差异 ===

架构：
  docker  → 客户端 + 守护进程（dockerd，root 运行）⭐
  podman  → 客户端直接调用 runc/crun（无守护进程）⭐⭐

root：
  docker  → 默认需要 root 或 docker 组（⚠️ 组等于 root）
  podman  → 原生支持 rootless ⭐⭐

pod：
  docker  → 无（用 compose 模拟）
  podman  → 原生 pod 概念（K8s 风格）⭐

systemd：
  docker  → 需要包装
  podman  → Quadlet / generate systemd 原生集成 ⭐⭐

兼容性：
  podman 兼容大部分 docker 命令（可 alias）⭐
  docker compose → 用 podman-compose 或 podman kube play

适用：
  docker  → 生态最全、文档最多、K8s 事实标准（containerd 除外）
  podman  → 安全要求高、无 root、RHEL 系默认 ⭐
EOF
```

#### 输出解读
`podman info` 的关键字段 ⭐：

| 字段 | 含义 |
|------|------|
| `host.security.rootless` | 是否 rootless ⭐⭐ |
| `host.cgroupVersion` | cgroup v1/v2 ⭐ |
| `store.graphRoot` | 镜像存储路径（rootless 在 `~/.local/share/containers`）⭐ |
| `host.ociRuntime.name` | 实际运行时（`crun`/`runc`）⭐ |
| `host.networkBackend` | `netavark`（默认）或 `cni` ⭐ |

**rootless 的 UID 映射** ⭐⭐：
```
宿主普通用户 (UID 1000)  ←→  容器内 root (UID 0)
宿主 subuid 范围          ←→  容器内其他 UID
```
由 `/etc/subuid` 与 `/etc/subgid` 决定映射范围 ⭐。

#### 常见坑
- ⚠️⚠️ **rootless 需要 `/etc/subuid` 与 `/etc/subgid` 配置**（缺了无法启动容器）⭐⭐：
  ```bash
  $ grep "$USER" /etc/subuid /etc/subgid
  user:100000:65536
  ```
- ⚠️⚠️ **rootless 容器不能绑定 1024 以下端口**（默认），除非设 `net.ipv4.ip_unprivileged_port_start` ⭐⭐：
  ```bash
  $ sudo sysctl -w net.ipv4.ip_unprivileged_port_start=80
  ```
- ⚠️ **`podman` 与 `docker` 的镜像存储是分开的**（不能直接互相使用），要用 `podman save`/`load` 或信任同一 registry ⭐。
- ⚠️ **`podman-compose` 与 `docker compose` 的兼容性不完全**（复杂配置会失败）⭐。
- ⚠️ **`podman generate systemd` 在 4.x 后标记为废弃**，推荐 Quadlet ⭐⭐。
- ⚠️ **rootless 下 `--network=host` 有额外限制**（仍受 unprivileged port 限制）⭐。
- **`podman` 官方名称是 `podman`（POD MANager）不是 `pod man`** ⭐。
- **容器名前面需要 `localhost/`**（podman 不默认用 docker.io 的路径解析）⭐：
  ```bash
  $ podman build -t myapp:1.0 .        # 本地镜像（localhost/myapp:1.0）
  ```
- 🔐 **rootless 显著降低「容器逃逸 → 宿主 root」的严重性**（攻击者拿到的是普通用户）⭐⭐。

#### 相关命令
`docker` · `buildah` · `skopeo` · `crun` · `podman-compose` · `kubectl`

### buildah / skopeo — 构建与搬运镜像

**常用度**：★★★☆☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：非 POSIX（Red Hat 工具）

```bash

```

| buildah 子命令 | 作用 |
|----------------|------|
| `from` | **从基础镜像创建容器** ⭐ |
| `run` | 在工作容器里执行命令 ⭐ |
| `copy` / `add` | 复制文件进容器 ⭐ |
| `config` | 设置元数据（ENTRYPOINT/CMD/ENV/USER） ⭐ |
| `commit` | **提交为镜像** ⭐⭐ |
| `bud` | 用 Dockerfile 构建（build-using-dockerfile）⭐ |
| `push` / `pull` | 推拉镜像 ⭐ |
| `images` / `rmi` | 镜像列表 / 删除 ⭐ |
| `mount` / `umount` | 挂载容器根文件系统 ⭐⭐ |
| `unshare` | 在 user namespace 里执行 ⭐ |

| skopeo 子命令 | 作用 |
|---------------|------|
| `inspect` | **检查远程镜像（不下载）** ⭐⭐ |

| `delete` | 删除远程标签 ⚠️ |
| `list-tags` | 列出标签 ⭐⭐ |
| `login` / `logout` | 登录 ⭐ |

```bash
$ sudo cp /tmp/binary "$mnt/usr/local/bin/"

$ buildah commit "$ctr" myapp:1.0
$ buildah rm "$ctr"

img=$1

skopeo inspect "docker-archive:$dst/$name.tar" | jq -r .Digest
skopeo inspect "docker://$img" | jq -r .Digest

echo "  skopeo copy docker-archive:$name.tar docker://registry.internal/myimg:tag"
```

#### 输出解读
`skopeo inspect` 的关键字段 ⭐⭐：

| 字段 | 含义 |
|------|------|
| `Name` | 完整镜像名 ⭐ |
| **`Digest`** | **内容哈希（不可变标识）** ⭐⭐ |
| `RepoTags` | 该 digest 对应的标签 ⭐ |
| `Created` / `DockerVersion` | 构建时间 / 工具版本 ⭐ |
| `Architecture` / `Os` | 目标平台 ⭐ |
| `Layers` | 层 digest 列表 ⭐ |
| `Env` | 环境变量（**可能含凭据**）🔐 |
| `Labels` | 标签（含 `org.opencontainers.image.*` 元数据）⭐ |

**镜像的四类存储格式（skopeo 之间可互转）** ⭐⭐：

| 格式 | 说明 |
|------|------|
| `docker://` | registry（远程） ⭐ |
| `dir:` | 解压的目录（便于检查）⭐ |
| `docker-archive:` | `docker save` 格式的 tar ⭐ |
| `oci:` | OCI 镜像布局（K8s/CRI 用）⭐ |
| `containers-storage:` | 本地容器存储（podman 的内部格式）⭐ |

**digest 与 tag 的差异** ⭐⭐：
```
tag（标签）    → 可变，可被重新指向别的 digest
digest（哈希） → 不可变，内容变了 digest 必变 ⭐⭐
```
⚠️ **生产部署用 digest 而不是 tag**（`latest` 尤其危险）⭐⭐。

#### 常见坑
- ⚠️⚠️ **用 tag 部署不可复现**（同一 tag 内容可能变）。**用 `@sha256:...` digest** ⭐⭐。
- ⚠️⚠️ **`skopeo inspect --config` 的 `.history[].created_by` 会暴露构建历史**（可能含硬编码凭据、内部 URL）🔐⭐⭐。
- ⚠️ **`skopeo copy` 默认只复制当前架构**，多架构要用 `--all` 或 `--multi-arch all` ⭐⭐。
- ⚠️ **`buildah run` 每次会创建层**（除非 `--volume` 挂载），大批量操作会堆积 ⭐。
- ⚠️ **`buildah` 与 docker 的镜像存储独立**，要 `buildah push` 到 registry 才能给 docker 用 ⭐。
- ⚠️ **rootless 下 `buildah` 需要 user namespace 支持**（`podman unshare`）⭐。
- **`skopeo inspect` 不下载镜像层**（只取 manifest），所以很快 ⭐⭐。
- **`skopeo` 不需要 docker 守护进程**（这是它相对 `docker pull` 的最大优势）⭐⭐。
- 🔐 **`skopeo` 是镜像供应链审计的必备工具**（不落地就能检查镜像元数据与历史）⭐⭐。

`podman` · `docker` · `crane`（go-containerregistry） · `trivy` · `cosign`

- **`nerdctl` 默认需要 root，但支持 rootless 模式**（`containerd-rootless-setuptool.sh`）⭐。
- **`nerdctl -n k8s.io` 是排查 Pod「容器起了但 Pod 没就绪」的好工具** ⭐⭐。

```bash
#!/bin/bash

for c in $(crictl ps -q); do
  priv=$(crictl inspect "$c" | jq -r '.info.runtimeSpec.linux.namespaces[]? | select(.type=="pid") | .path // "host?"')
```

| `ctr`/`nerdctl` | containerd（CRI 之下） | **运行时层面的排查** ⭐ |

⚠️ **`kubectl` 看不到的（如 Pause 容器、CRI 状态），`crictl` 能看到** ⭐⭐。

#### 常见坑
- ⚠️⚠️ **`crictl` 操作的是运行时状态，绕过 K8s 控制面**。删容器/Pod 后 **kubelet 会重建**，且可能与 K8s 状态不一致 🔐⭐。
- ⚠️⚠️ **`crictl rm`/`rmp` 不等于 `kubectl delete`**（不会更新 etcd 里的期望状态）🔐⭐。
- ⚠️ **必须配好 `runtime-endpoint`**（`/etc/crictl.yaml` 或 `-r`），否则报 `connection refused` ⭐⭐。
- ⚠️ **`crictl` 需要 root**（访问 containerd socket）⭐。
- ⚠️ **`crictl` 看不到 Pod 的 YAML 定义**（那是 K8s 层的），只能看运行时 spec ⭐。
- ⚠️ **`crictl logs` 与 `kubectl logs` 可能不一致**（前者是运行时的日志文件，后者可能经过 API 聚合）⭐。
- ⚠️ **`crictl ps` 默认只显示运行中的**，排查崩溃要 `-a` ⭐⭐。
- **`ATTEMPT` 列 > 0 就是 CrashLoopBackOff 的迹象**（配合 `kubectl describe pod` 看原因）⭐⭐。
- **`crictl info` 的 `.status.conditions` 是运行时健康状态**（`RuntimeReady`/`NetworkReady`）⭐⭐。
- 🔐 **`crictl` 能绕过 K8s 的准入控制与审计**做操作，是排查工具也是风险点 ⭐⭐。

#### 相关命令
`kubectl` · `ctr` · `nerdctl` · `containerd` · `kubelet` · `crictl`

**常用度**：★★★☆☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：非 POSIX（OCI 参考实现）

#### 输出解读
`runc state` 的关键字段 ⭐⭐：

| 字段 | 含义 |
|------|------|
| `id` | 容器 ID ⭐ |
| `pid` | **容器 init 进程在宿主上的 PID** ⭐⭐ |
| `status` | `created` / `running` / `paused` / `stopped` ⭐ |
| `bundle` | bundle 目录 ⭐ |
| `rootfs` | 根文件系统目录 ⭐ |
| **`namespace_paths`** | **各 namespace 的 `/proc/PID/ns/*` 路径** ⭐⭐ |

**OCI bundle 的组成** ⭐⭐：
```
bundle/
├── config.json        ← 容器配置（OCI spec）⭐⭐
└── rootfs/            ← 根文件系统 ⭐
```
⭐ **这就是「容器」的全部**：一份 JSON + 一个 rootfs + 内核的 namespace/cgroup。

**`config.json` 里最重要的安全字段** ⭐⭐：

| 字段 | 含义 |
|------|------|
| `process.capabilities.bounding` | 容器的 capability 上限 🔐⭐⭐ |
| `linux.namespaces[]` | 使用哪些 namespace（**缺失的项 = 与宿主共享**）🔐⭐⭐ |
| `linux.seccomp` | 系统调用过滤（`null` = 不过滤）🔐⭐ |
| `linux.maskedPaths` | 被屏蔽的路径（如 `/proc/kcore`）⭐ |
| `linux.readonlyPaths` | 只读路径（如 `/proc/sys`）⭐ |
| `root.readonly` | 根是否只读 ⭐ |
| `process.user.uid` | 运行 UID（`0` = root）⭐ |

#### 常见坑
- ⚠️⚠️ **`runc` 是低层工具，直接用它管理容器很危险**（无网络、无日志、无生命周期管理）。生产用 docker/podman/k8s ⭐⭐。
- ⚠️⚠️ **`config.json` 里 namespace 数组缺项 = 与宿主共享该 namespace**（这是 `--privileged` 的实质）🔐⭐⭐。
- ⚠️⚠️ **`runc spec` 生成的默认配置是「弱安全」的**（含较多 capability、无 seccomp）⭐⭐：
  ```bash
  $ jq '.process.capabilities.bounding | length' config.json
  14        # ⭐ docker 默认也是 ~14 个
  ```
- ⚠️ **`runc` 需要 root 才能启动容器**（除非配 rootless bundle）⭐。
- ⚠️ **`--no-pivot` 会让根切换不彻底**（容器可能看到宿主挂载点）🔐⭐。
- ⚠️ **`runc run` 是前台运行**（容器退出后命令才返回），要后台要 `-d` ⭐。
- ⚠️ **残留的容器状态会阻止同名容器创建**（要 `runc delete`）⭐。
- **`runc list -q` 只输出 ID**（脚本友好）⭐。
- **`runc` 的状态目录默认 `/run/runc`，重启丢失**（容器也不会自动恢复）⭐。
- 🔐 **`runc state` 的 `namespace_paths` 是「进入容器」的钥匙**（配合 `nsenter`）⭐⭐。

#### 相关命令
`nsenter` · `unshare` · `crictl` · `ctr` · `crun` · `oci-runtime-tool`

### unshare — 手动创建命名空间

**常用度**：★★★★☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：Linux 特有（`util-linux`）

```bash
unshare [选项] [命令 [参数...]]
```

| 选项 | 含义 |
|------|------|
| `--mount` / `-m` | **挂载 namespace**（需配合 `--propagation`） ⭐⭐ |
| `--uts` / `-u` | UTS（主机名/域名） ⭐⭐ |
| `--ipc` / `-i` | IPC（System V IPC/POSIX 消息队列） ⭐ |
| `--pid` / `-p` | **PID namespace（需 `--fork`）** ⭐⭐ |
| `--net` / `-n` | **网络 namespace（默认无网卡）** ⭐⭐ |
| `--user` / `-U` | user namespace（UID 映射） ⭐⭐ |
| `--cgroup` / `-C` | cgroup namespace ⭐ |
| `--time` / `-T` | time namespace（boottime/monotonic 偏移）⭐ |
| `--fork` / `-f` | **fork 子进程后在新 PID ns 里运行**（`--pid` 必须） ⭐⭐ |
| `--mount-proc[=目录]` | **在新 mount ns 里挂载新的 `/proc`**（`--pid` 配套） ⭐⭐ |
| `--map-root-user` | 把当前用户映射为容器内 root ⭐⭐ |
| `--map-user=UID` / `--map-group=GID` | 指定映射的 UID/GID ⭐ |
| `--propagation 模式` | 挂载传播（`private`/`slave`/`shared`） ⭐⭐ |
| `--setuid=UID` / `--setgid=GID` | 进入后切换 UID/GID ⭐ |
| `--keep-caps` | 保留 capability（配合 `--user`）⭐ |
| `--root=目录` | 切换根目录 ⭐ |
| `--wd=目录` | 工作目录 ⭐ |
| `--pid-file 文件` | 写 PID ⭐ |
| `--kill-child` / `--kill-child=SIG` | 外层退出时杀掉内层 ⭐ |

```bash
bash

$ cat > /tmp/tiny-container.sh <<'SH'

  --propagation private \
  bash -c '
    root="'"$root"'"
    hostname tiny
    ip link set lo up 2>/dev/null || true

    # 准备 chroot 里的挂载点
    mkdir -p "$root"/{proc,sys,dev,tmp}
    mount -t proc proc "$root/proc" 2>/dev/null || true
    mount --bind /dev "$root/dev" 2>/dev/null || true
    mount -t tmpfs tmpfs "$root/tmp"

    echo "=== 容器内信息 ==="
    echo "hostname: $(hostname)"
    echo "PID 1: $(cat /proc/1/comm)"
    echo "网络:"; ip -br a
    echo "挂载:"; mount | head -5
    echo "=== 进入 chroot ==="
    exec chroot "$root" '"$cmd"'
  '
SH
$ # ⭐ 需要先准备 rootfs（如用 debootstrap 或从镜像导出）
$ sudo mkdir -p /tmp/tinyroot && sudo debootstrap --variant=minbase bookworm /tmp/tinyroot \
>   http://deb.debian.org/debian 2>/dev/null | tail -2
$ sudo bash /tmp/tiny-container.sh /tmp/tinyroot /bin/bash
```

#### 输出解读
**`unshare` 后进程的 namespace**（`ls -l /proc/self/ns/`）⭐⭐：
```
total 0
lrwxrwxrwx 1 root root 0 ... cgroup -> cgroup:[4026531835]
lrwxrwxrwx 1 root root 0 ... ipc    -> ipc:[4026531839]
lrwxrwxrwx 1 root root 0 ... mnt    -> mnt:[4026531841]
lrwxrwxrwx 1 root root 0 ... net    -> net:[4026531840]
lrwxrwxrwx 1 root root 0 ... pid    -> pid:[4026531836]
lrwxrwxrwx 1 root root 0 ... user   -> user:[4026531837]
lrwxrwxrwx 1 root root 0 ... uts    -> uts:[4026531838]
```
⭐ **同编号 = 同一个 namespace**（无隔离）；**不同编号 = 已隔离** ⭐⭐。

**namespace 速查** ⭐⭐：

| namespace | 隔离内容 | 创建标志 |
|-----------|----------|----------|
| `mnt` | 挂载点 | `CLONE_NEWNS` ⭐ |
| `pid` | 进程 ID 空间 | `CLONE_NEWPID` ⭐ |
| `net` | 网络栈（接口/路由/iptables） | `CLONE_NEWNET` ⭐ |
| `ipc` | System V IPC、POSIX 队列 | `CLONE_NEWIPC` |
| `uts` | 主机名/域名 | `CLONE_NEWUTS` ⭐ |
| `user` | UID/GID 映射、capability | `CLONE_NEWUSER` ⭐⭐ |
| `cgroup` | cgroup 根视图 | `CLONE_NEWCGROUP` |
| `time` | `CLOCK_BOOTTIME`/`CLOCK_MONOTONIC` 偏移 | `CLONE_NEWTIME` （较新）⭐ |

⚠️ **`--pid` 必须配 `--fork` 与 `--mount-proc`**（否则看不到新 PID 空间，或 `/proc` 仍显示宿主的）⭐⭐。

#### 常见坑
- ⚠️⚠️ **`unshare --pid` 必须加 `--fork`**，否则 `unshare` 自己就在新 PID ns 里变成 PID 1 但后续命令看不到新空间 ⭐⭐。
- ⚠️⚠️ **`--pid` 不会自动重挂 `/proc`**，要 `--mount-proc`（否则 `/proc` 显示宿主的进程）⭐⭐。
- ⚠️⚠️ **`--mount` 后挂载可能影响宿主**（取决于 `--propagation`）。**显式 `--propagation private`** 更安全 ⭐⭐。
- ⚠️⚠️ **`--net` 创建的网络栈默认没有任何网卡（除 lo）与路由**，需要手动配置 veth/bridge ⭐⭐。
- ⚠️⚠️ **user namespace 曾被广泛用于提权漏洞（CVE 高发区）**。很多发行版收紧了非特权 userns 🔐⭐⭐：
  ```bash
  $ cat /proc/sys/kernel/unprivileged_userns_clone   # 0 = 禁止非特权创建
  ```
- ⚠️ **`--user` 与 `--map-root-user` 一起时，你要放弃真正的 root 权限**（只在新 ns 内是 root）⭐。
- ⚠️ **`--root` 会调用 `chroot`（比 `pivot_root` 弱）**，容器可能逃逸（`chroot` 逃逸是经典手法）🔐⭐。
- ⚠️ **`unshare` 需要相应的 capability**（`CAP_SYS_ADMIN`，或 user namespace 下的隐式能力）⭐。
- ⚠️ **`--mount-proc` 在不加 `--mount` 时会失败**（要先有 mount ns）⭐。
- 🔐 **`unshare` 是从零理解容器的最佳工具**（docker/podman 底层就是这些系统调用）⭐⭐。

#### 相关命令
`nsenter` · `chroot` · `runc` · `ip netns` · `systemd-nspawn` · `clone(2)`

### nsenter — 进入已有命名空间

**常用度**：★★★★★ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：Linux 特有（`util-linux`）

```bash
nsenter [选项] [命令 [参数...]]
```

| 选项 | 含义 |
|------|------|
| `-t PID` | **目标进程（从它的 namespace 进入）** ⭐⭐ |
| `-m` / `--mount[=文件]` | 进入 mount namespace ⭐⭐ |
| `-u` / `--uts` | UTS ⭐ |
| `-i` / `--ipc` | IPC ⭐ |
| `-n` / `--net` | **网络 namespace** ⭐⭐ |
| `-p` / `--pid` | **PID namespace** ⭐⭐ |
| `-U` / `--user` | user namespace ⭐ |
| `-C` / `--cgroup` | cgroup namespace ⭐ |
| `-T` / `--time` | time namespace ⭐ |
| `-a` / `--all` | **进入全部 namespace** ⭐⭐ |
| `-F` / `--no-fork` | 不 fork（默认会 fork）⭐ |
| `-f` / `--fork` | 显式 fork ⭐ |
| `-r` / `--root[=目录]` | 设置根目录 ⭐⭐ |
| `-w` / `--wd[=目录]` | 设置工作目录 ⭐⭐ |
| `-S UID` / `-G GID` | 切换 UID/GID ⭐ |
| `--preserve-credentials` | 不改变 UID/GID ⭐ |
| `--target PID` / `--net=文件` | 直接给 namespace 文件路径 ⭐ |

```bash
# 场景：⭐⭐⭐ 进入容器的全部 namespace（最常用的排障手段）⭐
$ pid=$(docker inspect -f '{{.State.Pid}}' web)

12345
$ sudo nsenter -t "$pid" -a                      # ⭐ 进入全部 ns（包括 / 与 cwd）
$ sudo nsenter -t "$pid" -a -- /bin/bash
$ # ⭐ 相当于 `docker exec`，但不需要容器里有 shell 之外的工具

# 场景：⭐⭐ 只进入网络 namespace（调网络最常用）⭐
$ sudo nsenter -t "$pid" -n ip -br a
$ sudo nsenter -t "$pid" -n ss -tunlp
$ sudo nsenter -t "$pid" -n tcpdump -i any -nn -c 20
$ sudo nsenter -t "$pid" -n iptables -L -n -v     # ⭐ 看容器内的 iptables
$ sudo nsenter -t "$pid" -n route -n

$ # ⭐ 合并宿主与容器的网络视图（在同一终端看两边）⭐
$ sudo nsenter -t "$pid" -n bash -c '
  echo "=== 容器网络 ==="; ip -br a; ip route; ss -tunlp
'
$ echo "=== 宿主网络 ==="; ip -br a; ss -tunlp | head

# 场景：⭐⭐ 从宿主直接查看/修改容器的文件系统 ⭐
$ sudo nsenter -t "$pid" -m -- ls -la /app
$ sudo nsenter -t "$pid" -m -w /app -- cat config.yml
$ sudo nsenter -t "$pid" -m -- find / -name '*.conf' -newermt '1 day ago' 2>/dev/null | head

# 场景：🔐⭐⭐ 排查「容器被入侵」的痕迹（不改容器）⭐
$ cat > /tmp/nsenter-ir.sh <<'SH'

  echo "hostname: $(hostname)"
  echo "PID 1: $(cat /proc/1/comm)"
  echo "内核: $(uname -r)"
  echo "cgroup: $(cat /proc/1/cgroup | head -2)"
'

echo
echo "=== 2. 进程列表（含隐藏检查）==="
sudo nsenter -t "$pid" -m -p -- ps -eo pid,ppid,user,stat,etime,cmd 2>/dev/null | head -30
echo "--- /proc 里的 PID 数 ---"
sudo nsenter -t "$pid" -m -p -- sh -c 'ls -d /proc/[0-9]* | wc -l'

echo
echo "=== 3. 可执行文件（找被替换/在临时目录的）==="
sudo nsenter -t "$pid" -m -- sh -c \
  'ls -l /proc/*/exe 2>/dev/null | grep -E "deleted|/tmp|/dev/shm" | head -20'

echo
echo "=== 4. 网络连接（找外连）==="
sudo nsenter -t "$pid" -n ss -tunap 2>/dev/null | head -30

echo
echo "=== 5. 被监听端口 ==="
sudo nsenter -t "$pid" -n ss -tunlp 2>/dev/null

echo
echo "=== 6. crontab / systemd / 启动文件 ==="
sudo nsenter -t "$pid" -m -- sh -c '
  for f in /etc/cron.d/* /etc/crontab /var/spool/cron/* /etc/rc.local; do

  find / -name authorized_keys 2>/dev/null | head -5

echo "=== 7. 最近改动的文件（24 小时内）==="
sudo nsenter -t "$pid" -m -- sh -c \
  'find / -xdev -type f -newermt "1 day ago" 2>/dev/null | grep -vE "^/proc|^/sys|^/tmp$" | head -30'

echo
echo "=== 8. 环境变量（可能含凭据）==="
sudo tr "\0" "\n" < "/proc/$pid/environ" | grep -iE "pass|token|key|secret" | head

echo
echo "=== 9. 挂载（找异常的宿主挂载）==="
sudo nsenter -t "$pid" -m -- mount | grep -vE "^proc|^sysfs|^tmpfs /(dev|sys|proc)|cgroup" | head -20

echo
echo "=== 10. 是否特权 ==="
sudo grep CapEff /proc/"$pid"/status
sudo capsh --decode="$(sudo awk "/CapEff/{print \$2}" /proc/"$pid"/status)" 2>/dev/null | tr "," "\n" | head -20
SH
$ sudo bash /tmp/nsenter-ir.sh "$pid"

# 场景：⭐⭐ 排查「容器 OOM / 卡死」（进不去时用）⭐
$ # 1. 找出容器 PID
$ docker inspect -f '{{.State.Pid}}' app
$ # 2. 即使容器卡死，也能从宿主进入看状态

$ sudo cat /sys/fs/cgroup/"$(cat /proc/"$pid"/cgroup | head -1 | cut -d: -f3)"/memory.current

# 场景：🔐⭐⭐ 用 nsenter 检查容器的「真实网络视图」（不是 docker inspect）
$ sudo nsenter -t "$pid" -n bash -c '
  echo "=== 接口 ==="; ip -br a
  echo "=== 路由 ==="; ip route
  echo "=== ARP ==="; ip neigh
  echo "=== 监听 ==="; ss -tunlp
  echo "=== 连接 ==="; ss -tunap state established
  echo "=== DNS ==="; cat /etc/resolv.conf
  echo "=== 防火墙 ==="; iptables -L -n 2>/dev/null | head -20

# 场景：⭐ 直接操作容器内的进程（宿主视角）⭐
$ sudo nsenter -t "$pid" -p -m -- kill -TERM 1        # ⚠️ 相当于停止容器
$ sudo nsenter -t "$pid" -p -- ps -o pid,ppid,stat,cmd -p 1
```

#### 输出解读
`nsenter` 的核心逻辑 ⭐⭐：
```
nsenter -t <目标的 PID> <要进入的 ns> <命令>
         │              │              └─ 在新 ns 里执行（默认 fork）
         │              └─ 可多个；-a 表示全部
         └─ 从该进程「复制」namespace 引用
```

⚠️ **进入 namespace 后，进程的 `/proc/PID/ns/*` 指向新 ns**，但**PID 号不变**（除非进入 PID ns 并 fork）⭐。

**`-a` 包含的 namespace** ⭐⭐：
```
-m (mount) -u (uts) -i (ipc) -n (net) -p (pid) -C (cgroup) -U (user) -T (time)
```

**`-r` 与 `-w` 的配合** ⭐⭐：
```bash
$ nsenter -t PID -m -r -w /app -- /bin/sh
#            │  │  └─ 工作目录设为 /app（在新 mount ns 里）
#            │  └─ 根目录设为目标的根（相当于 chroot 到容器 rootfs）
#            └─ 先进入 mount ns（否则看不到容器的 rootfs）
```
⚠️ **`-r` 需要先 `-m`**（否则看不到容器的根文件系统）⭐。

#### 🔐 `nsenter` 的安全意义
| 用途 | 说明 |
|------|------|
| **绕过容器内限制** | 从宿主进入，不受容器内 SELinux/seccomp/工具缺失限制 ⭐⭐ |
| **应急响应** | 容器卡死/被入侵时从宿主取证 ⭐⭐ |
| **调试** | 容器没有 shell 或缺少工具时 ⭐⭐ |
| **风险** | 攻击者拿到宿主 root 后可用它进入任意容器 ⚠️ |
| **风险** | 特权容器 + `nsenter -t 1 -a` 可进入宿主的 namespace ⚠️⚠️ |

**特权容器 + PID ns 的致命组合** ⭐⭐：
```bash
# 若容器用了 --pid=host，容器内可以直接进入宿主的所有 namespace
$ nsenter -t 1 -a           # ⚠️⚠️ 进入 PID 1（宿主 init）的全部 namespace
# → 相当于完全逃逸到宿主
```
🔐 **这是最经典的容器逃逸路径之一**，排查时要检查 `--pid=host` 与 `CAP_SYS_ADMIN`。

#### 常见坑
- ⚠️⚠️ **`nsenter -t PID` 需要目标进程存在**（容器退出后 PID 消失，就进不去了）⭐。
- ⚠️⚠️ **需要 root 或与目标进程有足够的权限**（`CAP_SYS_ADMIN`/`CAP_SYS_PTRACE`）⭐⭐。
- ⚠️⚠️ **`-a` 进入 PID namespace 时会 fork**，你在新 ns 里是 PID 1 的子进程；**退出后回到原 ns** ⭐。
- ⚠️⚠️ **`-r`（setns 到 root）不是 `pivot_root`**，安全性弱于真正的容器根切换（可能被 chroot 逃逸）🔐⭐。
- ⚠️ **在容器内部用 `nsenter` 不一定工作**（需要 `CAP_SYS_ADMIN` 且目标在同 PID ns）⭐。
- ⚠️ **`nsenter` 不改变 PID**（只是进入 PID ns），所以 `ps` 看到的 PID 是「新 ns 里的编号」⭐。
- **`nsenter -t PID -a -- <命令>` 是 `docker exec` 的底层实现**（docker exec 也走类似路径）⭐⭐。
- **容器卡死时 `docker exec` 会失败，但 `nsenter` 常能成功**（因为不经过 dockerd）⭐⭐。
- **`nsenter --net=/proc/PID/ns/net` 可在不指定 PID 的情况下进网络 ns** ⭐。
- 🔐 **`nsenter` 是「绕过容器隔离」的标准手法**，攻防两端都要掌握 ⭐⭐。

#### 相关命令
`unshare` · `chroot` · `runc` · `ip netns` · `docker exec` · `setns(2)`

### chroot — 切换根目录

**常用度**：★★★★☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：POSIX 定义了 `chroot(2)`；GNU 的 `chroot` 命令有 `--userspec`/`--groups` 扩展

```bash
chroot [选项] 新根 [命令 [参数...]]
```

| 选项 | 含义 |
|------|------|
| `--userspec=用户:组` | 切换 UID/GID（可写名字或数字） ⭐ |
| `--groups=组列表` | 设置附加组 ⭐ |
| `--skip-chdir` | 不切到新根的 `/` ⭐ |

```bash
# 场景：⭐⭐ 基本用法（准备救援/调试环境）
$ sudo mkdir -p /mnt/rescue/{proc,sys,dev,dev/pts,tmp,etc}
$ sudo chroot /mnt/rescue /bin/bash
$ # ⚠️ 新根里必须有 bash 及其依赖的库，否则报 "No such file or directory"

$ # ⭐ 用 busybox 构造最小环境
$ sudo mkdir -p /tmp/miniroot/{bin,proc,sys,dev,tmp}
$ sudo cp /bin/busybox /tmp/miniroot/bin/
$ sudo chroot /tmp/miniroot /bin/busybox sh
/ # ls /
bin  dev  proc  sys  tmp

# 场景：⭐⭐ 救援模式：修复系统（最实际的用途）⭐
$ cat > /tmp/rescue-chroot.sh <<'SH'
#!/bin/bash
set -euo pipefail
root=${1:-/mnt/sysroot}
[ -d "$root" ] || { echo "根目录不存在: $root"; exit 1; }

echo "=== 1. 挂载必要的伪文件系统 ==="
for fs in proc sys dev; do
  mountpoint -q "$root/$fs" || mount --rbind "/$fs" "$root/$fs"
done
[ -d "$root/dev/pts" ] && mountpoint -q "$root/dev/pts" || mount --rbind /dev/pts "$root/dev/pts"
mountpoint -q "$root/run" || mount --rbind /run "$root/run" 2>/dev/null || true

echo "=== 2. 确认是有效的系统 ==="
[ -x "$root/bin/bash" ] || [ -x "$root/usr/bin/bash" ] || { echo "⚠️ 找不到 bash"; exit 1; }
ls "$root/etc/os-release" >/dev/null 2>&1 && cat "$root/etc/os-release" | grep PRETTY_NAME

echo "=== 3. 进入 chroot（退出后执行清理）==="
chroot "$root" /bin/bash || true

echo "=== 4. 清理挂载 ==="
umount -R "$root/dev/pts" 2>/dev/null || true
for fs in proc sys dev run; do
  umount -R "$root/$fs" 2>/dev/null || true
done
echo "完成"
SH
$ # 示例：从 Live USB 修复原系统
$ sudo mount /dev/nvme0n1p2 /mnt/sysroot
$ sudo mount /dev/nvme0n1p1 /mnt/sysroot/boot
$ sudo bash /tmp/rescue-chroot.sh /mnt/sysroot
# 在 chroot 里可以：
#   passwd root                    # 重置密码
#   vim /etc/fstab                 # 修 fstab
#   grub-mkconfig -o /boot/grub/grub.cfg
#   pacman -Syu / apt install ...  # 重装包
#   systemctl enable sshd

# 场景：🔐⭐ 让 chroot 更「容器化」（隔离更多）⭐
$ # ⭐ chroot 默认只改根目录，其他都不隔离
$ cat > /tmp/safer-chroot.sh <<'SH'
#!/bin/bash
# 用 unshare 包一层，让 chroot 也隔离 UTS/IPC/MOUNT
set -euo pipefail
root=${1:?用法: $0 <新根> [命令]}
shift
cmd=${*:-/bin/bash}

sudo unshare --mount --uts --ipc --propagation private \
  chroot "$root" "$cmd"
# ⭐ 再进一步（含 PID 与网络）：
#   sudo unshare --mount --uts --ipc --pid --fork --net --mount-proc \
#     chroot "$root" "$cmd"
SH
$ sudo bash /tmp/safer-chroot.sh /tmp/miniroot /bin/busybox sh

$ # ⭐ 指定用户与组（不必以 root 进入）
$ sudo chroot --userspec=1000:1000 /mnt/rescue /bin/bash
$ sudo chroot --userspec=nobody:nogroup /tmp/miniroot /bin/busybox id
$ sudo chroot --groups=1000,1001 /tmp/miniroot /bin/busybox id

# 场景：🔐⭐⭐ chroot 逃逸（经典技术，必须理解以便防守）⭐⭐
$ cat > /tmp/chroot-escape-demo.sh <<'SH'
#!/bin/bash
# ⚠️ 演示：仅限自有环境！说明「chroot 不是安全边界」
set -uo pipefail
root=/tmp/escaperoot

=== chroot vs namespace 容器 ===

chroot 只做一件事：
  改变进程看到的「/」在哪（路径解析起点）⭐
  ✅ 简单、POSIX、任何 Unix 都有
  ❌ 只隔离路径：进程仍能看到宿主进程、用宿主网络、
     有 root 时还能逃逸（chroot 逃逸是经典手法）⚠️⚠️

namespace 容器做了更多：
  ✅ 隔离 PID/NET/MNT/UTS/IPC/USER/CGROUP（看到的世界是独立的）
  ✅ 可限制资源（cgroup）
  ✅ 可限制 capability（丢掉大部分 root 权限）
  ✅ 可过滤系统调用（seccomp）

正确的 chroot 用法：
  - 救援模式修系统（宿主 root 主动进入）⭐
  - 干净构建/测试环境（不追求安全隔离）⭐
  - 作为「最小容器」的一层（配合 unshare + cap 限制）⭐

错误的用法：
  ❌ 用 chroot 当安全沙箱运行不可信代码 ⚠️⚠️
EOF
```

#### 输出解读
`chroot` 的行为 ⭐：
```
chroot(2) 做的事：
  1. 把「进程的根目录」指向新路径
  2. 改变工作目录（默认切到新根的 /）
  3. 之后所有绝对路径都从新根开始解析
```
⚠️ **不做的事**：不改 UID、不隔离网络/PID、不限制 capability ⭐⭐。

**`/proc/PID/root` 是判断「进程的根在哪里」的权威方式** ⭐⭐：
```bash
$ sudo ls -l /proc/PID/root
lrwxrwxrwx ... /proc/PID/root -> /tmp/miniroot      # ⭐ 在 chroot 里
$ readlink /proc/PID/root
```

**chroot 逃逸的原理** ⭐⭐：
```
前提：在 chroot 内有 root 权限 且保留 CAP_SYS_CHROOT

步骤：
1. mkdir /tmp/escape
2. chroot /tmp/escape     ← 现在“根”变成 /tmp/escape
3. cd ../../../../..      ← 一直往上（此时可以走到原根的上级！）
4. chroot .               ← 把“根”改回去
5. 现在能看到并访问真正的宿主 /

防御：
  ✅ 进入 chroot 前 drop 掉 CAP_SYS_CHROOT
  ✅ 用 pivot_root 而不是 chroot
  ✅ 用 namespace（mount ns + pivot_root）
  ✅ 用 seccomp 禁掉 chroot 系统调用
```

#### 🔐 为什么 chroot 不是安全边界
| 维度 | chroot 隔离 | 容器隔离 |
|------|-------------|----------|
| 路径 | ✅ | ✅ |
| 进程可见 | ❌（`ps` 看得到全部） | ✅（PID ns） |
| 网络 | ❌（共用宿主网络栈） | ✅（NET ns） |
| 主机名 | ❌ | ✅（UTS ns） |
| capability | ❌（仍是 root 的全部） | ✅（可 drop） |
| 系统调用 | ❌ | ✅（seccomp） |
| 资源 | ❌ | ✅（cgroup） |
| 逃逸难度 | **低（有 root 即可）** ⚠️⚠️ | 取决于配置 |

#### 常见坑
- ⚠️⚠️ **`chroot` 不是安全边界**（有 root 即可逃逸，且不隔离 PID/网络）🔐⭐⭐。
- ⚠️⚠️ **新根里缺少动态库时会报 `No such file or directory`**（实际上缺的是 `ld-linux`）⭐⭐：
  ```bash
  $ sudo chroot /tmp/miniroot /bin/bash
  chroot: failed to run command '/bin/bash': No such file or directory   # ⚠️ 其实是缺库
  $ ldd /bin/bash      # ⭐ 把列出的库都拷进新根
  ```
- ⚠️ **`chroot` 后 `/proc` 需要手动挂载**（否则 `ps` 等工具不工作）⭐。
- ⚠️ **`chroot` 后 `/dev` 需要挂载或创建必要设备节点**（如 `/dev/null`、`/dev/urandom`）⭐：
  ```bash
  $ sudo mount --rbind /dev /tmp/miniroot/dev     # ⭐ 最省事
  ```
- ⚠️ **退出 chroot 前要卸载里面的挂载**（否则 `umount` 外层目录会失败）⭐⭐。
- ⚠️ **`chroot` 不改变 UID**（仍是进入前的用户，通常是 root）⭐。
- ⚠️ **在 chroot 里 `systemctl` 通常不能工作**（除非挂载了 `/run` 与 cgroup）⭐。
- **`chroot` 是救援模式的必备工具**（重置密码、修 fstab、重装引导器）⭐⭐。
- **`--userspec` 可在 chroot 里降权**（`nobody`）⭐。
- 🔐 **审计要点**：`ps` 里的 chroot 进程、`/tmp` 里带 `bin/`+`lib/` 的目录、SUID 的 mount/chroot 类工具 ⭐⭐。

#### 相关命令
`unshare` · `nsenter` · `pivot_root` · `bubblewrap` · `systemd-nspawn` · `debootstrap`

### systemd-nspawn — 轻量容器

**常用度**：★★★☆☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：Linux 特有（systemd）

```bash
systemd-nspawn [选项] [命令]
systemd-nspawn -D 目录
```

| 选项 | 含义 |
|------|------|
| `-D 目录` / `--directory=` | **用该目录当容器根** ⭐⭐ |
| `-b` / `--boot` | **在容器里启动 systemd（完整容器）** ⭐⭐ |
| `--machine=名字` | 容器名（配合 `machinectl`） ⭐ |
| `-M 名字` | 同 `--machine` ⭐ |
| `--private-users=yes` | **隔离 UID（user namespace）** ⭐⭐ |
| `--private-network` | 无网络（只有 lo） ⭐ |
| `--network-veth` | 创建 veth 对（连到宿主） ⭐⭐ |
| `--network-bridge=br0` | 接到某网桥 ⭐ |
| `--port=宿主端口:容器端口` | 端口转发 ⭐ |
| `--bind=源[:目标][:ro]` | **绑定挂载** ⭐⭐ |
| `--bind-ro=` | 只读绑定 ⭐ |
| `--tmpfs=路径` | 挂载 tmpfs ⭐ |
| `--read-only` | 根只读 ⭐ |
| `--capability=列表` | 设置 capability ⭐🔐 |
| `-u 用户` / `--user=` | 以某用户运行 ⭐ |
| `--settings=yes/no` | 应用 `systemd.nspawn` 配置 ⭐ |
| `-P` / `--pipe` | 管道模式（命令用管道而非终端） ⭐ |
| `-q` / `--quiet` | 安静 ⭐ |
| `--volatile=yes|state|overlay` | 临时根（改动不落盘） ⭐⭐ |

```bash
# 场景：⭐⭐ 用现成 rootfs 启动容器
$ sudo mkdir -p /var/lib/machines/debian12
$ sudo debootstrap --include=systemd,dbus,iproute2 bookworm /var/lib/machines/debian12 \
>   http://deb.debian.org/debian
$ sudo systemd-nspawn -D /var/lib/machines/debian12 /bin/bash
Spawning container debian12 on /var/lib/machines/debian12.
# ⭐ 这是一个轻量容器（不启动 systemd，只是 chroot+namespace）

$ # ⭐ 完整容器（容器内跑 systemd）
$ sudo systemd-nspawn -D /var/lib/machines/debian12 -b --machine=debian12
$ # 在另一个终端：
$ sudo machinectl list
$ sudo machinectl shell debian12
$ sudo systemctl -M debian12 status
$ sudo journalctl -M debian12 -n 20

# 场景：⭐⭐ 网络配置
$ # 无网络
$ sudo systemd-nspawn -D /var/lib/machines/debian12 --private-network -b
$ # veth（自动分配地址，NAT 到宿主）
$ sudo systemd-nspawn -D /var/lib/machines/debian12 --network-veth -b
$ ip -br a | grep -E 've-|vz-'            # ⭐ 宿主的 veth 端
$ sudo machinectl shell debian12 ip -br a
$ # 接到网桥
$ sudo systemd-nspawn -D /var/lib/machines/debian12 --network-bridge=br0 -b
$ # ⭐ 端口转发（veth + 宿主的 NAT 转发）
$ sudo systemd-nspawn -D /var/lib/machines/debian12 --network-veth --port=8080:80 -b

# 场景：⭐⭐ 绑定挂载与只读根
$ sudo systemd-nspawn -D /var/lib/machines/debian12 \
>   --bind=/srv/data:/data --bind-ro=/etc/app.conf:/etc/app.conf \
>   --tmpfs=/tmp --read-only \
>   /bin/bash
$ # ⭐ 在容器里：/data 可写、/etc/app.conf 只读、根只读

# 场景：🔐⭐⭐ 临时容器（改动不落盘，安全的实验环境）⭐
$ sudo systemd-nspawn -D /var/lib/machines/debian12 --volatile=yes -b
$ sudo systemd-nspawn -D /var/lib/machines/debian12 --ephemeral /bin/bash
$ # ⭐ --ephemeral：用 overlay 快照，退出后一切改动消失
$ # ⭐ 适合：测试不可信软件、验证配置、跑危险脚本

# 场景：🔐 安全加固选项
$ sudo systemd-nspawn -D /var/lib/machines/debian12 -b \
>   --private-users=yes \                       # ⭐ UID 隔离
>   --private-network \                          # ⭐ 无网络
>   --capability=CAP_NET_BIND_SERVICE \          # ⭐ 最小 capability
>   --read-only \                                # ⭐ 根只读
>   --system-call-filter='~@clock @debug @module' \  # ⭐ seccomp

$ sudo machinectl start debian12            # ⭐ 用 machinectl 管理（读取同一配置）

$ cat > /tmp/nspawn-audit.sh <<'SH'
#!/bin/bash
for m in $(machinectl list --no-legend 2>/dev/null | awk '{print $1}'); do
  echo "=== $m ==="
  mpid=$(systemctl show -p MainPID --value "machine-$m.scope" 2>/dev/null)
  [ -n "$mpid" ] && [ "$mpid" != "0" ] && {
    echo "--- namespace"
    for ns in /proc/"$mpid"/ns/*; do
      t=$(readlink "$ns"); p=$(readlink "/proc/1/ns/$(basename "$ns")")

    capsh --decode="$(awk '/CapEff/{print $2}' /proc/"$mpid"/status)" 2>/dev/null | tr ',' '\n' | head -8
```

#### 输出解读
`systemd-nspawn` 与 docker 的关系 ⭐：

| 维度 | systemd-nspawn | docker |
|------|----------------|--------|
| 守护进程 | **无**（调用即进程）⭐ | dockerd |
| 设计目标 | **像轻量 VM 一样跑一个完整 OS** ⭐⭐ | 跑单个应用 |
| 容器内 init | **完整 systemd**（`-b`）⭐⭐ | 通常单进程 |
| 镜像 | 一个目录（rootfs）⭐ | OCI 镜像 |
| 管理 | `machinectl` + `systemd-nspawn@` unit ⭐ | docker CLI |
| 网络 | `--network-veth`/`--network-bridge` ⭐ | 自建 bridge |
| 端口转发 | `--port=8080:80` ⭐ | `-p` |

**`--volatile` 的三种模式** ⭐⭐：

| 值 | 行为 |
|----|------|
| `yes` | `/usr` 只读 + 其他目录放 tmpfs（**改动全丢**） ⭐ |
| `state` | `/var` 放 tmpfs（保留系统，只丢状态） ⭐ |
| `overlay` | 用 overlayfs 做可写层（改动不落盘） ⭐⭐ |

**`.nspawn` 配置的四个段** ⭐：

| 段 | 内容 |
|----|------|
| `[Exec]` | 启动方式、Private、Capability、SystemCallFilter ⭐ |
| `[Network]` | VirtualEthernet、Bridge、Port ⭐ |
| `[Files]` | Bind、BindReadOnly、TemporaryFileSystem ⭐ |
| `[Security]` | NoNewPrivileges、ProtectSystem 等 ⭐ |

#### 常见坑
- ⚠️⚠️ **nspawn 容器共享宿主内核**（不是 VM），隔离强度取决于选项 ⭐⭐。
- ⚠️⚠️ **不加 `--private-users` 时容器内的 root 就是宿主的 root**（危险）🔐⭐⭐。
- ⚠️⚠️ **不加 `-b` 时容器内不跑 init**（只是 namespace + chroot），服务无法用 systemctl ⭐。
- ⚠️ **`--network-veth` 需要宿主有可用的 NAT（IP 转发）**，否则容器出不了网 ⭐⭐：
  ```bash
  $ sudo sysctl -w net.ipv4.ip_forward=1
  $ sudo iptables -t nat -A POSTROUTING -s 10.x.x.x/24 -o eth0 -j MASQUERADE
  ```
- ⚠️ **容器 rootfs 里要有 systemd 与 dbus**（否则 `-b` 失败）⭐。
- ⚠️ **`--volatile=yes` 会丢 `/var`（含日志）**，排查问题时不方便 ⭐。
- ⚠️ **`machinectl` 需要 systemd 与 root** ⭐。
- **`systemd-nspawn` 最适合「在一个隔离环境里跑完整发行版」**（比 docker 更像 VM，比 VM 更轻）⭐⭐。
- **`--ephemeral` 是「安全的实验环境」**（改动自动丢弃）⭐⭐。

`machinectl` · `systemd-nspawn@.service` · `docker` · `podman` · `unshare` · `debootstrap`

#### 输出解读
`kubectl get pods` 的列 ⭐⭐：

| 列 | 含义 |
|----|------|
| `NAME` | Pod 名 ⭐ |
| `READY` | `就绪容器数/总容器数`（**`1/2` 说明有容器未就绪**）⭐⭐ |
| `STATUS` | `Running`/`Pending`/`CrashLoopBackOff`/`ImagePullBackOff`/`Completed`/`Error` ⭐⭐ |
| `RESTARTS` | **重启次数（持续增长 = 不稳定）** ⭐⭐ |
| `AGE` | 存活时长 ⭐ |

**常见 STATUS 与原因** ⭐⭐：

| 状态 | 常见原因 |
|------|----------|
| `Pending` | 资源不足、PVC 未绑定、节点选择器不匹配、污点 ❌ |
| `ContainerCreating` | 拉镜像/挂卷中（可能卡住） |
| `CrashLoopBackOff` | **容器启动后马上退出（看 `--previous` 日志）** ⭐⭐ |
| `ImagePullBackOff` | 镜像拉取失败（名字错/私有仓库未配置凭据）⭐⭐ |
| `ErrImagePull` | 同上（首次失败的状态）⭐ |
| `OOMKilled` | 内存超 limits 被杀（看 `describe` 的 Last State）⭐⭐ |
| `Error` | 容器退出码非 0 ⭐ |
| `Completed` | Job/CronJob 正常结束 ⭐ |
| `Terminating` | 正在删除（可能卡在 finalizer）⭐ |
| `Evicted` | 节点资源压力被驱逐 ⭐ |

**`describe` 里最该看的两段** ⭐⭐：
```
Last State:     Terminated
  Reason:       OOMKilled          ← ⭐ 上次为什么退出
  Exit Code:    137                ← ⭐ 128+9（SIGKILL）

Events:                            ← ⭐⭐ 最关键
  Type     Reason     Age   From     Message
  ----     ------     ----  ----     -------
  Warning  Failed     2m    kubelet  Failed to pull image "nginx:wrong": ...
  Normal   Pulled     1m    kubelet  Successfully pulled image
  Warning  BackOff    30s   kubelet  Back-off restarting failed container
```

**退出码速查** ⭐：`137` = SIGKILL（常见 OOM）、`143` = SIGTERM、`1` = 应用错误、`126`/`127` = 无法执行/找不到命令。

#### 🔐 K8s 安全排查要点
| 检查项 | 命令 |
|--------|------|
| 特权 Pod | `-o json \| jq ...privileged==true` ⭐⭐ |
| hostNetwork/PID/IPC | `-o json \| jq ...hostPID` ⭐🔐 |
| hostPath 挂载 `/` 或 `docker.sock` | `-o json \| jq ...hostPath.path` ⭐🔐 |
| cluster-admin 绑定 | `get clusterrolebinding` ⭐ |
| 默认 SA 被滥用 | `-o json \| jq ...serviceAccountName=="default"` ⭐ |
| Secret 可读性 | `get secrets -A` + `auth can-i get secrets` ⭐ |
| 匿名访问 | `auth can-i get pods --as=system:anonymous` ⭐🔐 |
| RBAC 中的 `*` 权限 | `get clusterrole -o json \| jq '...rules[].verbs=="*"'` ⭐ |

#### 常见坑
- ⚠️⚠️ **不加 `-n` 或 `-A` 时只看当前命名空间**（默认 `default`），会误以为「资源不存在」⭐⭐。
- ⚠️⚠️ **`kubectl logs` 只显示当前容器实例**；崩溃过的要用 `--previous` ⭐⭐。
- ⚠️⚠️ **`kubectl apply` 与 `create` 的差异**：前者幂等可更新，后者重复执行会报错。**用 `apply`** ⭐⭐。
- ⚠️⚠️ **`kubectl delete pod` 后 Pod 会被控制器重建**（Deployment 管理时），要改副本数或 Deployment ⭐。
- ⚠️ **`CrashLoopBackOff` 的退避时间指数增长**（10s → 20s → 40s ... 最长 5 分钟），排查时可能「等很久才看到日志」⭐。
- ⚠️ **`port-forward` 是临时隧道**（本地调试用，不是生产暴露方式）⭐。
- ⚠️ **`kubectl run` 在新版本上默认创建 Pod 而不是 Deployment**（旧版是 Deployment）⭐。
- ⚠️ **`-o yaml` 的输出不可直接回贴 apply**（含 `status`、`resourceVersion`、`uid` 等只读字段）⭐⭐：
  ```bash
  $ kubectl get deploy web -o yaml | grep -v 'creationTimestamp: null' > clean.yml
  $ # ⭐ 或用 `kubectl get -o yaml | kubectl neat`
  ```
- **`kubectl diff` 在 CUPS 版本支持后非常好用**（提交前预览变更）⭐⭐。
- 🔐 **`~/.kube/config` 常含集群管理员凭据**，等同于集群权限 🔐⭐⭐。
- 🔐 **`kubectl exec` 进入容器可以绕过容器的日志/审计**（是排查工具，也是审计盲区）⭐。

**Helm 的三个核心概念** ⭐⭐：
```
chart      = 模板 + 默认值（可复用的「安装包」）⭐
values     = 覆盖模板的值（你的配置）⭐⭐
release    = chart + values 在集群里的一次实例化 ⭐⭐
```

**values 的覆盖优先级**（从低到高）⭐⭐：
```
1. chart 自带 values.yaml
2. 父 chart 的 values.yaml（子 chart 场景）
3. -f 指定的文件（可多次，后面的覆盖前面）
4. --set / --set-string / --set-file
```
⭐ **`helm get values --all` 能看到完整的最终值**。

**Helm 3 的存储** ⭐⭐：
```
release 数据 → K8s Secret（名字 sh.helm.release.v1.<release>.v<N>）
               │
               └─ ⚠️ 内含 values（可能明文凭据）与 manifest
```
⚠️ **能读 Secret 的人 = 能看到所有 values** 🔐⭐⭐。

#### 常见坑
- ⚠️⚠️ **`--set` 的密码会写进 release 的 Secret（可被有权限者读到）** 🔐⭐⭐：
  ```bash
  $ # ✅ 正确：用 K8s Secret
  $ kubectl create secret generic x --from-literal=pass=...
  $ helm install app chart --set auth.existingSecret=x
  ```
- ⚠️⚠️ **第三方 chart 可能默认部署特权容器/hostPath**。**部署前必须 `helm template` + `helm show values`** 🔐⭐⭐。
- ⚠️⚠️ **`helm uninstall` 默认保留 PVC（数据）**，但**不保留 release 历史**。要审计用 `--keep-history` ⭐⭐。
- ⚠️⚠️ **`helm upgrade` 不加 `--atomic` 时失败会留下半成品**（部分资源更新、部分没）⭐⭐。
- ⚠️ **`helm rollback` 只回滚 K8s 资源，不回滚数据/PVC** ⭐⭐。
- ⚠️ **`--set` 的语法对特殊字符很敏感**（逗号、点、等号要转义）⭐：
  ```bash
  $ helm install x c --set 'a.b\.c=1'          # ⭐ 转义点
  $ helm install x c --set 'list={a,b,c}'      # ⚠️ 逗号含义不同
  $ helm install x c --set-file key=./file.txt # ⭐ 从文件读值
  ```
- ⚠️ **`helm get values` 默认只显示「用户覆盖的值」**，不是完整值（要 `--all`）⭐⭐。
- ⚠️ **chart 版本与应用版本是两回事**（`--version` 指定 chart，`appVersion` 是应用）⭐。
- **`helm template` 是「零风险审查」的最佳工具**（不发到集群）⭐⭐。
- **Helm 3 不再需要 Tiller**（Helm 2 的 Tiller 是高权限组件，已被移除）⭐。
- 🔐 **`helm.sh/hook` 的 pre-install 会在集群里执行 Job**（可能执行任意命令）🔐⭐。

#### 常见坑
- ⚠️⚠️ **kind 里用本地构建的镜像必须 `kind load docker-image`**，否则 `ImagePullBackOff` ⭐⭐。
- ⚠️⚠️ **minikube/kind 的 kubeconfig 会写入 `~/.kube/config` 并切换 context**，容易误操作到错误的集群 ⭐⭐：
  ```bash
  $ kubectl config current-context      # ⭐ 每次操作前确认！
  $ kubectl config get-contexts
  ```
- ⚠️⚠️ **k3s 的 `/etc/rancher/k3s/k3s.yaml` 含集群管理员凭据**（`chmod 600`，不要 `644` 暴露给所有用户）🔐⭐⭐。
- ⚠️ **kind 的集群在 docker 容器里，删除 kind 集群不会删 docker 镜像**（要单独清理）⭐。
- ⚠️ **`minikube delete` 会删除集群与数据**（PVC 数据丢失）⭐。
- ⚠️ **kind 的 `extraPortMappings` 需要节点容器启动时指定**（不能后加）⭐⭐。
- ⚠️ **本地集群默认无 NetworkPolicy、无 PodSecurity 限制**（生产前必须补）🔐⭐。
- **`kind get kubeconfig --name X` 可导出配置**（不污染主 kubeconfig）⭐：
  ```bash
  $ kind get kubeconfig --name lab > /tmp/lab.kubeconfig
  $ KUBECONFIG=/tmp/lab.kubeconfig kubectl get nodes
  ```
- **k3s 适合「一台机器上的 K8s」，kind 适合「CI 里的临时集群」** ⭐⭐。

### vagrant — 虚拟机声明式编排

**常用度**：★★★☆☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：无（Ruby 程序，跨平台）

`vagrant` 用一份 `Vagrantfile`（Ruby DSL）描述"用哪个 box、给多少内存、装什么、映射什么端口"，一条 `up` 就把环境拉起来。它本身**不虚拟化**，只是驱动 VirtualBox / libvirt / VMware / Hyper-V 的"遥控器"。学习场的价值：一条命令复现"3 台 CentOS 互 ping"的实验拓扑，且 `destroy` 能干净回滚。

```bash
vagrant init -m generic/centos9s                    # -m 生成最小 Vagrantfile（不加则生成满是注释的长模板）
vagrant up                                          # 创建并启动（首次会下载 box）
vagrant status                                      # 查看各机器状态
vagrant ssh web01                                   # SSH 进多机 Vagrantfile 中的 web01
vagrant halt web01                                  # 关机（保留磁盘）
vagrant suspend / resume web01                      # 挂起 / 恢复
vagrant reload --provision                          # 重启并重跑 provision 脚本
vagrant provision                                   # 只重跑配置脚本
vagrant box list                                    # 本机已下载的 box
vagrant box update                                  # 更新 box（不会自动改已建机器）
vagrant destroy -f                                  # ⚠️ 删除所有机器与磁盘，不可恢复（未保存的数据全丢）
vagrant snapshot save clean && vagrant snapshot restore clean   # 快照回滚（仅 VirtualBox/libvirt）
```

多机拓扑的最小 Vagrantfile：

```ruby
Vagrant.configure("2") do |config|
  config.vm.box = "generic/centos9s"
  # 全局关闭默认的 /vagrant 共享目录（避免每个 VM 都挂一份宿主目录）
  config.vm.synced_folder ".", "/vagrant", disabled: true

  config.vm.define "web01" do |web|
    web.vm.hostname = "web01"
    web.vm.network "private_network", ip: "192.168.56.11"
    web.vm.provider "virtualbox" do |vb|
      vb.memory = 1024
      vb.cpus   = 1
    end
    web.vm.provision "shell", inline: <<-SHELL
      dnf install -y epel-release || true
      echo "web01 ready" > /etc/motd
    SHELL
  end

  config.vm.define "db01" do |db|
    db.vm.hostname = "db01"
    db.vm.network "private_network", ip: "192.168.56.12"
  end
end
```

```bash
$ vagrant up

==> web01: Importing base box 'generic/centos9s'...
==> web01: Running provisioner: shell...
    web01: web01 ready
$ vagrant status
Current machine states:
web01                     running (virtualbox)
db01                      not created (virtualbox)
```

**输出解读**

| 状态 | 含义 |
|------|------|
| `not created` | 只有定义，磁盘还没建 |
| `running` | 已启动 |
| `poweroff` | 磁盘存在但关机（`up` 会很快恢复） |
| `saved` | 挂起状态（快照在磁盘上） |
| `aborted` | 进程被强杀，状态可能不一致，需 `reload` |

**常见坑**

- ⚠️ `vagrant destroy -f` **不询问**，且会删除虚拟磁盘。做实验前先确认数据都在宿主或共享目录里。
- `vagrant ssh` 用的是 box 内置的**不安全私钥**（Vagrant 官方 insecure key，密码就是 `vagrant`）。**绝不要**把带 private_network 或 host-only 网卡的 Vagrant 机器暴露到公网。
- 私有网络 `private_network` 默认网段 `192.168.56.0/24` 在 VirtualBox 里叫 host-only 网络；如果宿主已有同名网卡冲突，地址会下发失败但 `up` 仍报成功 —— 进机器 `ip a` 确认。
- `config.vm.synced_folder` 默认把当前目录挂到 `/vagrant`。在多机环境里每台都挂一份，宿主的密钥可能被所有机器读到 🔐。
- 改完 Vagrantfile **必须** `vagrant reload`，`up` 对已运行的机器不会重建网卡 / 内存配置。
- `provision` 脚本**默认只跑一次**（首次 `up`）。想每次开机都跑加 `run: "always"`。

**相关命令**：`docker compose` · `virsh` · `VBoxManage` · `multipass` · `limactl`

### virsh / virt-install — KVM 与 libvirt 管理

**常用度**：★★★★☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：无（Linux 专有，依赖 libvirt 守护进程）

libvirt 把 QEMU/KVM、Xen、LXC 统一成一套 API，`virsh` 是它的命令行客户端，`virt-install` 是"创建虚拟机"的封装。运维和渗透里 `virsh` 高频出现的两个原因：① 排障（看宿主机上跑着哪些 VM、VM 的网络/磁盘在哪）；② 逃逸与取证（VM 磁盘是 qcow2 文件，宿主机 root 可直接挂载读取 🔐）。

```bash
virsh list --all                                    # 列出所有域（含关机），--all 必加否则只显示运行中
virsh dominfo web01                                 # 单个 VM 的 vCPU / 内存 / UUID
virsh start web01 / shutdown web01 / destroy web01  # 启动 / 优雅关机 / ⚠️ 断电式强制终止
virsh reboot web01                                  # 重启
virsh console web01                                 # 串口控制台（Ctrl+] 退出）
virsh define /etc/libvirt/qemu/web01.xml            # 从 XML 定义（不启动）
virsh edit web01                                    # 编辑 XML（必须用它，直接改文件会被覆盖）
virsh dumpxml web01                                 # 导出当前生效的 XML
virsh domblklist web01                              # 该 VM 的磁盘路径列表
virsh domiflist web01                               # 网卡与所属虚拟网络
virsh iface-list / net-list --all                   # 宿主物理网卡 / 虚拟网络
virsh net-dumpxml default                           # 默认 NAT 网络的网段与 DHCP 池
virsh snapshot-create-as web01 clean "干净基线"      # 建快照
virsh snapshot-list web01
virsh snapshot-revert web01 clean                   # 回滚
virsh pool-list --all / vol-list default            # 存储池
virsh autostart web01 / autostart --disable web01   # 开机自启
```

`virt-install` 一条命令建机（适合脚本化、课堂批量分发）：

```bash
sudo virt-install \
  --name web01 --memory 2048 --vcpus 2 \
  --disk path=/var/lib/libvirt/images/web01.qcow2,size=20,format=qcow2 \
  --cdrom /iso/CentOS-Stream-9-latest-x86_64-dvd1.iso \
  --network network=default,model=virtio \
  --graphics none --console pty,target_type=serial \
  --os-variant centos-stream9
```

```bash
$ virsh list --all
 Id   Name    State
-------------------------
 1    web01   running
 -    db01    shut off
$ virsh domblklist web01
 Target   Source
-----------------------------------------------
 vda      /var/lib/libvirt/images/web01.qcow2
$ virsh dumpxml web01 | grep -A2 '<interface'
    <interface type='network'>
      <source network='default'/>
```

**输出解读**

| 字段 / 状态 | 含义 |
|------|------|
| `Id` 为 `-` | 关机状态（运行中的 Id 是正整数） |
| `running` / `shut off` / `paused` / `crashed` | 域状态；`paused` 常见于宿主 I/O 卡死或手工 pause |
| `<source network='default'>` | 走 libvirt 的 `virbr0` NAT，VM 可出网但外部进不来 |
| `<source bridge='br0'>` | 桥接，VM 和宿主同级、有真实 IP |
| `Target vda` | virtio 磁盘，性能最好；`sda` 是 SATA/SCSI 模拟 |

**常见坑**

- 直接 `vi /etc/libvirt/qemu/web01.xml` 改配置**无效**，libvirt 运行时会用内存里的定义覆盖文件。必须 `virsh edit`。
- `virsh destroy` 相当于**拔电源**，不是关机。常规用 `shutdown`（发 ACPI），无响应再 `destroy`。
- `<disk type='file'>` 里的 qcow2 用 `qemu-img info` 能看到 backing file（差分镜像）。如果宿主上存在他人可读的 backing 文件，等于所有差分层数据可被还原 🔐。
- VM 的内存可被宿主 root 通过 `/proc/<qemu-pid>/mem` 或内存 dump 读取，`virsh dump --memory-only` 能导出完整内存（取证利器，也是逃逸后的取证点）。
- 默认网络 `default` 只在宿主重启后**不自动启动**（老版本）；`virsh net-autostart default` 固化。
- 不要给 VM 挂 `--hostdev` 直通宿主块设备，除非你清楚这条路径能让 VM 写坏宿主磁盘。

**相关命令**：`qemu-img` · `qemu-system-x86_64` · `virsh` · `docker` · `VBoxManage`

### qemu-img / qemu-system-x86_64 — 磁盘镜像与全虚拟化

**常用度**：★★★★☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：无

`qemu-img` 是**离线的镜像工具**（查看 / 转换 / 建盘 / 快照，不启动虚拟机），`qemu-system-x86_64` 是**全系统模拟器**（既能用 KVM 加速，也能纯软件模拟别种架构）。在安全场景里 `qemu-img` 的出现频率极高：把 VM 磁盘转成 raw 挂到宿主分析、从镜像里提取文件、做取证快照 🔐。

```bash
qemu-img info disk.qcow2                            # 看虚拟大小 / 实际占用 / backing file
qemu-img create -f qcow2 -o size=20G disk.qcow2     # 建空盘
qemu-img create -f qcow2 -b base.qcow2 -F qcow2 diff.qcow2   # 基于 base 建差分盘（写时复制）
qemu-img convert -p -O raw disk.qcow2 disk.raw      # ⭐ 转 raw 后可 mount / 用 strings 分析
qemu-img convert -p -O qcow2 disk.raw disk.qcow2    # 反向压缩
qemu-img resize disk.qcow2 +10G                     # ⚠️ 只扩容器，不扩内部文件系统，需再 resize2fs
qemu-img check disk.qcow2                           # 校验镜像一致性（出错先备份再修）
qemu-img snapshot -c clean disk.qcow2               # 打内部快照
qemu-img snapshot -l disk.qcow2
qemu-img snapshot -a clean disk.qcow2               # 回滚到快照
qemu-img rebase -u -b base.qcow2 -F qcow2 diff.qcow2 # 安全改写 backing 路径
qemu-img map --output=json disk.qcow2               # 看哪些块被真正写入（判断稀疏）
qemu-img compare a.raw b.raw                        # 逐块比对两个镜像
```

直接启动一个 QEMU 虚拟机（无 libvirt，纯命令行）：

```bash
# 用 KVM 加速启动，带 VNC 供远程看画面
sudo qemu-system-x86_64 \
  -enable-kvm -cpu host -smp 2 -m 2048 \
  -drive file=disk.qcow2,if=virtio,format=qcow2 \
  -netdev user,id=n0,hostfwd=tcp::2222-:22 \
  -device virtio-net-pci,netdev=n0 \
  -vnc 127.0.0.1:1 -daemonize

# 纯软件模拟其他架构（无需 KVM，慢但可跑 ARM/MIPS 固件）
qemu-system-aarch64 -M virt -cpu cortex-a57 -m 1024 \
  -kernel vmlinuz -append "console=ttyAMA0" -nographic
```

```bash
$ qemu-img info disk.qcow2
image: disk.qcow2
file format: qcow2
virtual size: 20 GiB (21474836480 bytes)
disk size: 1.9 GiB
cluster_size: 65536
backing file: base.qcow2 (actual path: /var/lib/libvirt/images/base.qcow2)
Snapshot list:
ID   TAG    VM SIZE   DATE        VM CLOCK
1    clean  0 B       2026-09-15  00:00:12.345
```

**输出解读**

| 字段 | 含义 |
|------|------|
| `virtual size` | VM 看到的盘大小（元数据里声明的） |
| `disk size` | 宿主上**实际占用**；远小于 virtual 说明是稀疏盘 |
| `backing file` | 差分盘的父层，链式依赖 |
| `Snapshot list` 的 `VM SIZE` 为 `0 B` | 该快照与当前状态无内存快照，只回滚磁盘 |
| `-netdev user,hostfwd=` | 用户态 NAT，`hostfwd` 做端口转发（宿主 2222 → VM 22） |

用 `-netdev user` 时 VM 访问不到宿主以外的网络细节，渗透实验要用 `-netdev tap` + 桥接。

**常见坑**

- ⚠️ `qemu-img resize` 只改**容器的虚拟大小**，分区表与文件系统不会自动扩展。扩完还要进系统 `growpart` + `resize2fs` / `xfs_growfs`，否则 `df` 毫无变化。
- ⚠️ 差分盘**删不得父层**。删了 `base.qcow2`，`diff.qcow2` 立刻报废（`qemu-img check` 会报 `backing file not found`）。
- 转 raw 会按 **virtual size** 落盘（不是 disk size）。20G 虚拟盘转 raw 就是实打实 20G，先确认宿主空间。
- 回滚快照会**丢弃快照之后的所有写入**，先确认没有未保存数据。
- `qemu-img` 对**正在被 VM 使用**的镜像操作是危险的（可能读到不一致状态）。分析运行中的 VM 磁盘，要么用 `virsh snapshot`，要么先关机，要么用只读方式：`qemu-img convert -O raw` 副本。
- 取证时对原始镜像一律 `-r`（只读）打开，避免 `qemu-img check` 的自动修复写入证据 🔐。
- `-enable-kvm` 在嵌套虚拟化不可用时静默退回 TCG（纯软件），慢几十倍；确认 `/dev/kvm` 存在且当前用户有权限。

**相关命令**：`virsh` · `virt-install` · `mount -o loop,ro` · `dd` · `losetup`

### VBoxManage — VirtualBox 命令行

**常用度**：★★★☆☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：无

VirtualBox 的图形界面在无头服务器上没用，`VBoxManage` 是它的完整命令行等价物。和 libvirt 不同，它自成一派，配置存在 `~/.config/VirtualBox/VirtualBox.xml` 与每台机器的 `.vbox` 文件里。

```bash
VBoxManage list vms                                 # 所有 VM 名称与 UUID
VBoxManage list runningvms
VBoxManage list ostypes | grep -i ubuntu            # 支持的 guest 类型 ID
VBoxManage createvm --name web01 --ostype Ubuntu_64 --register
VBoxManage modifyvm web01 --memory 2048 --cpus 2 --nic1 nat
VBoxManage modifyvm web01 --natpf1 "ssh,tcp,,2222,,22"   # 端口转发（宿主2222→guest22）
VBoxManage modifyvm web01 --nic2 hostonly --hostonlyadapter2 vboxnet0
VBoxManage storagectl web01 --name "SATA" --add sata --controller IntelAhci
VBoxManage storageattach web01 --storagectl "SATA" --port 0 --device 0 \
  --type hdd --medium /data/web01.vdi
VBoxManage createmedium disk --filename web01.vdi --size 20480          # 单位 MB
VBoxManage clonemedium web01.vdi web01-clone.vdi --format VDI
VBoxManage startvm web01 --type headless            # 无头启动（服务器必备）
VBoxManage controlvm web01 poweroff                 # ⚠️ 断电；acpipowerbutton 才是优雅关机
VBoxManage controlvm web01 acpipowerbutton
VBoxManage snapshot web01 take clean
VBoxManage snapshot web01 restore clean
VBoxManage showvminfo web01 --machinereadable       # 机器可读格式，适合脚本
VBoxManage guestcontrol web01 run --exe /bin/id --username root --password pw --wait-stdout
VBoxManage list hostonlyifs
VBoxManage modifyvm web01 --natdnshostresolver1 on  # 让 guest 用宿主 DNS
VBoxManage list systemproperties | grep 'Default machine'
VBoxManage export web01 -o web01.ova               # 导出 OVA（可搬迁）
VBoxManage import web01.ova
```

```bash
$ VBoxManage list runningvms
"web01" {3f2c8a4e-1d5b-4a6f-9c21-7d8e0f1a2b3c}
$ VBoxManage showvminfo web01 | grep -iE 'Memory size|Number of CPUs|NIC 1'
Memory size                  2048MB
Number of CPUs:              2
NIC 1:                       MAC: 080027A1B2C3, Attachment: NAT, Cable connected: on
```

**输出解读**

| 字段 | 含义 |
|------|------|
| `{UUID}` | 虚拟机唯一标识；同名 VM 靠 UUID 区分，脚本应用 UUID 而非名字 |
| `Attachment: NAT` | NAT 模式，出网易、入网需 `--natpf` 端口转发 |
| `Attachment: Host-only` | 只与宿主互通，适合隔离实验网 |
| `Attachment: Bridged` | 桥接到物理网卡，VM 拿真实 IP，**等于接入你的局域网** 🔐 |
| `--type headless` | 不弹窗口；服务器上必须加，否则报错退出 |

**常见坑**

- `controlvm poweroff` 是**拔电**，文件系统可能损坏；日常用 `acpipowerbutton`。
- `Bridged` 网卡会让 VM 和你的笔记本在**同一广播域**。实验用的靶机（含漏洞镜像）接 Bridged 就等于接在办公网里，务必用 Host-only / NAT 🔐。
- `createmedium --size` 单位是 **MB**（不是 GB），写 20 会建出 20MB 的盘，装系统时才发现。
- `modifyvm` 必须在**关机状态**改；运行中改内存/网卡要用 `controlvm`（且只有部分参数支持）。
- `guestcontrol` 需要在 guest 里装 Guest Additions 且知道账号密码；不理解原理就授权，等于把宿主的执行能力递进 guest。
- 快照会占用和**磁盘同量级**的空间（差分盘），多个快照叠加速磁盘膨胀，跑实验定期 `snapshot delete`。

**相关命令**：`vagrant` · `virsh` · `qemu-img` · `docker`

### distrobox / toolbox — 在宿主里跑发行版容器

**常用度**：★★★★☆ ｜ **所属**：容器与虚拟化 ｜ **POSIX/GNU 差异**：无（需 Podman 或 Docker）

`distrobox` 用容器把"另一个发行版"装进来，但**深度集成宿主**：自动挂载 `$HOME`、`/tmp`、`$XAUTHORITY`，能用宿主的声音、显示、USB。Arch 用户想用 apt 装的工具、或想在一个干净环境里试搭环境，这是最省事的路子 —— 比 `chroot` 方便，比虚拟机快。

```bash
distrobox create --name ubu --image ubuntu:24.04          # 建一个 Ubuntu 容器
distrobox create --name kali --image kalilinux/kali-rolling --home ~/kali-home  # 独立 HOME，隔离配置
distrobox list                                             # 列出所有盒子
distrobox enter ubu                                        # 进入（首次会初始化）
distrobox enter ubu -- sudo apt update                     # 直接执行一条命令后退出
distrobox export --app firefox                             # 把容器的 GUI 程序"导出"到宿主菜单
distrobox export --bin /usr/bin/nmap --name nmap-ubu       # 只导出命令
distrobox upgrade --all / distrobox upgrade ubu             # 升级容器内软件
distrobox stop ubu / distrobox rm ubu                       # 停止 / 删除（--force 跳过确认）
distrobox assemble create --file distrobox.ini              # 从清单批量建（团队/课堂复现环境）
```

```bash
# distrobox.ini —— 一条命令拉起整套实验环境
[ubuntu-dev]
image=ubuntu:24.04
additional_packages="build-essential,git,gdb,python3-pip"
init_hooks="echo 'export PS1=\"[ubu] \w\$ \"' >> ~/.bashrc"

[kali-tools]
image=kalilinux/kali-rolling
additional_packages="nmap,netcat-openbsd,curl"
home=~/kali-home
```

```bash
$ distrobox list
ID           NAME        STATUS             IMAGE
1            ubu         Up 2 hours         ubuntu:24.04
2            kali        Exited             kalilinux/kali-rolling
$ distrobox enter ubu -- cat /etc/os-release | head -2
NAME="Ubuntu"
VERSION="24.04.1 LTS (Noble Numbat)"
```

**输出解读**

| 字段 | 含义 |
|------|------|
| `Up` / `Exited` | 容器状态；`Exited` 时 `enter` 会自动拉起 |
| `home=~/kali-home` | 容器内 HOME 指向宿主目录，**配置持久化**但会污染宿主 dotfiles |
| `export --app` | 在宿主 `~/.local/share/applications/` 生成 .desktop，菜单里直接出现 |
| `additional_packages` | 建容器时一次性安装；改清单后要 `assemble create --force` |

**常见坑**

- 默认容器与宿主**共享 `$HOME`**：容器里 `rm` 的东西宿主也没有了。装"会乱改配置"的软件（渗透工具集）时用 `--home ~/xxx` 隔离。
- `distrobox` 不是沙箱 🔐。为了让 GUI 和 USB 能用，它给容器开了大量宿主权限（接近 `--privileged` 的一小部分能力）。**不要**在 distrobox 里跑不可信代码并当它隔离。
- 内部用户 UID 与宿主一致（rootless Podman 映射），所以容器里能读写宿主文件，权限模型是"宿主用户"而不是"隔离的 root"。
- 依赖 Podman/Docker 存在。Arch 上装 `distrobox` 会拉一堆 Podman 依赖，纯为试一个包可能不划算。
- `distrobox rm` 不加 `--force` 会交互确认；脚本里要显式加。
- `toolbox` 是 Fedora 官方的同类工具，面向 Silverblue/不可变系统，功能比 distrobox 保守（默认不共享整个 HOME 到容器 root）。

**相关命令**：`podman` · `docker` · `chroot` · `systemd-nspawn`
